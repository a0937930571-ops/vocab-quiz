        let state = {};


        function initQuiz() {

            const container = document.getElementById('quiz-container');

            container.innerHTML = '';


            questionsData.forEach(q => {

                state[q.id] = { answered: false, correct: false };


                const card = document.createElement('div');

                card.className = 'question-card';

                card.id = `card-${q.id}`;


                const sentenceWithBlank = q.text.replace('_____', `<span class="blank" id="blank-${q.id}">______</span>`);


                card.innerHTML = `

                    <div class="q-header">

                        <span class="q-number">Question ${q.id}</span>

                        <button class="redo-btn" onclick="resetQuestion(${q.id})" title="重作此題">↻</button>

                    </div>

                    <div class="sentence">${sentenceWithBlank}</div>

                    <div class="options-grid" id="options-${q.id}">

                        ${Object.entries(q.options).map(([key, value]) => `

                            <button class="option-btn" id="btn-${q.id}-${key}" onclick="checkAnswer(${q.id}, '${key}')">

                                (${key}) ${value}

                            </button>

                        `).join('')}

                    </div>

                `;

                container.appendChild(card);

            });

            updateHeader();

        }


        function checkAnswer(qId, key) {

            if (state[qId].answered) return;


            const q = questionsData.find(item => item.id === qId);

            const isCorrect = key === q.ans;


            state[qId].answered = true;

            state[qId].correct = isCorrect;


            const blankEl = document.getElementById(`blank-${qId}`);

            blankEl.innerText = key;


            const selectedBtn = document.getElementById(`btn-${qId}-${key}`);

            const correctBtn = document.getElementById(`btn-${qId}-${q.ans}`);


            if (isCorrect) {

                selectedBtn.classList.add('correct');

            } else {

                selectedBtn.classList.add('wrong');

                correctBtn.classList.add('correct');

            }


            Object.keys(q.options).forEach(optKey => {

                document.getElementById(`btn-${qId}-${optKey}`).disabled = true;

            });


            updateHeader();

        }


        function resetQuestion(qId) {

            if (!state[qId].answered) return;


            state[qId].answered = false;

            state[qId].correct = false;


            const blankEl = document.getElementById(`blank-${qId}`);

            blankEl.innerText = "______";


            const q = questionsData.find(item => item.id === qId);

            Object.keys(q.options).forEach(optKey => {

                const btn = document.getElementById(`btn-${qId}-${optKey}`);

                btn.disabled = false;

                btn.classList.remove('correct', 'wrong');

            });


            updateHeader();

        }


        function updateHeader() {

            const completedCount = Object.values(state).filter(s => s.answered).length;

            const scoreCount = Object.values(state).filter(s => s.correct).length;

            const total = questionsData.length;


            document.getElementById('completed').innerText = completedCount;

            document.getElementById('score').innerText = scoreCount;

            document.getElementById('progress').style.width = `${(completedCount / total) * 100}%`;

        }


        window.onload = initQuiz;
